import { supabase } from '@/lib/supabase';

// Slovak classifieds · Next.js + Supabase
export async function getLatestListings() {
  const { data } = await supabase
    .from('listings')
    .select('id, title, price, created_at')
    .order('created_at', {
      ascending: false,
    });
  return data ?? [];
}

public class SailBoat : NetworkBehaviour {
  [Networked] float Heading { get; set; }
  WindField wind = WindField.Uniform();

  // Real-time wind physics · PhysX
  public override void FixedUpdateNetwork() {
    var force = wind.ForceOn(sail, Heading);
    rb.AddForceAtPosition(force,
      rb.worldCenterOfMass);
  }
}

              I'm a software engineer who completed a BSc in Software Engineering at the University of

              August 2026. The VH Software work continues.
            

            

              I'm comfortable across the stack: Flutter on the client, Firebase on the backend, Unity for
              simulation, Next.js and Supabase when the job calls for it, and now enterprise frontend
              with TypeScript and Angular in a banking context. I like problems that sit at the
              intersection of engineering precision and product thinking.
